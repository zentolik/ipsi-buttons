    function readPayload() {
        var h = location.hash || '';
        var marker = '#cb=';
        var i = h.indexOf(marker);
        if (i < 0) return null;
        var raw = h.slice(i + marker.length);
        try { return JSON.parse(decodeURIComponent(raw)); }
        catch (e) { try { return JSON.parse(raw); } catch (e2) { return null; } }
    }

    function resolveBrand(data) {
        var b = (data.project_brand || data.client_brand || '').trim();
        var comp = (data.project_company || '').toLowerCase();
        if (comp.indexOf('schweiz') >= 0 || comp.indexOf('switzerland') >= 0) {
            if (b === 'United Media') return 'United Media Schweiz';
        }
        if (GROUP_MAP[b]) return b;
        var keys = Object.keys(GROUP_MAP);
        for (var i = 0; i < keys.length; i++) {
            if (b.toLowerCase().indexOf(keys[i].toLowerCase()) >= 0) return keys[i];
        }
        return 'Euroweb';
    }

    function pickVisible(selector) {
        var all = document.querySelectorAll(selector);
        for (var i = 0; i < all.length; i++) { if (all[i].offsetParent !== null) return all[i]; }
        return null;
    }
    function waitForElm(selector, timeout) {
        timeout = timeout || 15000;
        return new Promise(function (resolve, reject) {
            var found = pickVisible(selector);
            if (found) return resolve(found);
            var to = setTimeout(function () { obs.disconnect(); reject(new Error('timeout ' + selector)); }, timeout);
            var obs = new MutationObserver(function () {
                var el = pickVisible(selector);
                if (el) { clearTimeout(to); obs.disconnect(); resolve(el); }
            });
            obs.observe(document.body, { childList: true, subtree: true });
        });
    }
    function setNativeValue(el, value) {
        var proto = Object.getPrototypeOf(el);
        var desc = Object.getOwnPropertyDescriptor(proto, 'value');
        if (desc && desc.set) { desc.set.call(el, value); } else { el.value = value; }
        el.dispatchEvent(new Event('input', { bubbles: true }));
        el.dispatchEvent(new Event('change', { bubbles: true }));
    }
    function delay(ms) { return new Promise(function (r) { setTimeout(r, ms); }); }
    function escapeHtml(s) { return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }

    /* ===== 4) FIELD SETTERS ===== */
    function switchTab(type) {
        var tab = pickVisible('.tab[data-type="' + type + '"]');
        if (tab) { tab.click(); return true; }
        return false;
    }
    async function fillTitle(title) {
        var el = await waitForElm('input[name="title"]');
        setNativeValue(el, title);
    }
    async function fillKundennummer(nr) {
        var el = pickVisible('input[name="kundennummer"]');
        if (el) setNativeValue(el, nr || '');
    }
    async function fillBody(content) {
        var el = await waitForElm('div[contenteditable="true"][data-name="body"], div[data-name="body"][contenteditable="true"], div[contenteditable="true"]');
        el.focus();
        if (content && typeof content === "object" && content.html != null) {
            el.innerHTML = content.html;
        } else {
            el.innerHTML = String(content).split("\n").map(function (l) { return escapeHtml(l); }).join("<br>");
        }
        el.dispatchEvent(new Event("input", { bubbles: true }));
        el.dispatchEvent(new Event("keyup", { bubbles: true }));
        el.dispatchEvent(new Event("blur", { bubbles: true }));
    }
    async function setSelectByName(name, value) {
        var el = pickVisible('select[name="' + name + '"]');
        if (el) setNativeValue(el, String(value));

    async function setGroupById(id) {
        var shadow = pickVisible('input.searchableSelect-shadow[name="group_id"]');
        if (!shadow) { await delay(300); shadow = pickVisible('input.searchableSelect-shadow[name="group_id"]'); }
        if (!shadow) return false;
        var wrap = shadow.closest('.searchableSelect') || shadow.parentElement;
        var main = wrap.querySelector('input.searchableSelect-main');
        if (main) { main.focus(); main.click(); }
        await delay(200);
        var li = wrap.querySelector('li[data-value="' + id + '"]');
        if (!li) {
            var lis = document.querySelectorAll('li[data-value="' + id + '"]');
            for (var i = 0; i < lis.length; i++) { if (lis[i].offsetParent !== null) { li = lis[i]; break; } }
        }
        if (!li) return false;
        li.dispatchEvent(new MouseEvent('mousedown', { bubbles: true }));
        li.dispatchEvent(new MouseEvent('mouseup', { bubbles: true }));
        li.click();
        await delay(150);
        if (String(shadow.value) !== String(id)) setNativeValue(shadow, String(id));
        return true;
    }

    async function fillCustomer(email) {
        var input = await waitForElm('input[name="customer_id_completion"]');

        input.dispatchEvent(new KeyboardEvent('keydown', { bubbles: true, key: 'a' }));

            list = pickVisible('ul.recipientList.is-shown') || document.querySelector('ul.recipientList.is-shown');
            if (list && list.querySelectorAll('li.recipientList-entry').length) break;
            await delay(150);
        }
        if (!list) return;
        var entries = list.querySelectorAll('li.recipientList-entry.js-object[data-object-id]');
        var match = null;
        for (var i = 0; i < entries.length; i++) {
            var txt = (entries[i].textContent || '').toLowerCase();
            if (email && txt.indexOf(email.toLowerCase()) >= 0) { match = entries[i]; break; }
        }
        if (!match && entries.length === 1) match = entries[0];
        if (match) {
            match.dispatchEvent(new MouseEvent('mousedown', { bubbles: true }));
            match.click();
        }
        // no match: leave typed email, NEVER click li.recipientList--new
    }

    async function runZammad() {
        var data = readPayload();
        if (!data || !data.action_type) return;
        try { await waitForElm('input[name="title"]', 20000); } catch (e) { return; }
        await delay(400);

        var kdnr = data.client_id || data.kundennummer || '';
        var brand = resolveBrand(data);
        var groups = GROUP_MAP[brand] || GROUP_MAP['Euroweb'];
        var at = data.action_type;

        if (at === 'domaintransfer' || at === 'yourrate' || at === 'dritttermin_vl') {
            // INTERNAL -> phone-in, own email
            switchTab('phone-in');
            await delay(300);
            await fillKundennummer(kdnr);
            await fillCustomer(ownEmail(data));
            if (at === 'domaintransfer') {
                await setGroupById(groups.onlineschaltung);
                await fillTitle(subjectDomaintransfer(kdnr, data.web_hunter));
                await fillBody(TPL.domaintransfer_body(kdnr, data.web_hunter));
            } else if (at === 'yourrate') {
                await setGroupById(groups.yourrate);
                await fillTitle(subjectYourRate(kdnr));
                await fillBody(TPL.yourrate_body);
                showYourRatePopups(data);
            } else {
                await setGroupById(groups.dritttermin);
                await fillTitle(subjectVL(kdnr));
                await fillBody(TPL.vl_body(data.client_name || '', kdnr, data.demo_link));
            }
            await setSelectByName('state_id', STATE.offen);
            await setSelectByName('priority_id', 3);
        } else {
            // EXTERNAL customer mail -> email-out, gender popup
            switchTab('email-out');
            await delay(300);
            var anrede = await askGender(data.client_name || '');
            await fillKundennummer(kdnr);
            await fillCustomer(data.client_email || '');
            await setGroupById(data.is_aktu ? groups.aktu : groups.web);
            await fillTitle((kdnr || '') + ' – ' + ((data.project_type || 'Anfrage').trim()));
            var nm = (data.client_name || '').trim();
            var greet = anrede || 'Guten Tag';
            var line = (greet.indexOf('Guten Tag') < 0 && nm) ? (greet + ' ' + nm) : greet;
            await fillBody(line + ',\n\n');
            await setSelectByName('state_id', STATE.offen);
            await setSelectByName('priority_id', 3);
        }
    }

    /* ===== 10) DOKU WEB-HUNTER BRIDGE (runs inside doku archive tab) ===== */
    function isDoku() { return location.hostname.indexOf('doku.') >= 0; }

    function readWebHunter() {
        var table = document.querySelector('table');
        if (!table) return null;
        var headers = Array.prototype.map.call(table.querySelectorAll('thead th, thead td'), function (h) { return h.textContent.trim(); });
        var apptIdx = headers.findIndex(function (h) { return /appointment type/i.test(h); });
        var hunterIdx = headers.findIndex(function (h) { return /web hunter/i.test(h); });
        var subIdx = headers.findIndex(function (h) { return /submitted/i.test(h); });
        if (apptIdx < 0 || hunterIdx < 0) return null;
        function parseDate(s) {
            var m = (s || '').match(/(\d{2})\/(\d{2})\/(\d{4}),\s*(\d{2}):(\d{2})\s*([ap])\.m\./i);
            if (!m) { var m2 = (s || '').match(/(\d{2})[.\/-](\d{2})[.\/-](\d{4})/); return m2 ? new Date(+m2[3], +m2[2] - 1, +m2[1]).getTime() : 0; }
            var h = parseInt(m[4], 10);
            if (/p/i.test(m[6]) && h !== 12) h += 12;
            if (/a/i.test(m[6]) && h === 12) h = 0;
            return new Date(+m[3], +m[1] - 1, +m[2], h, +m[5]).getTime();
        }
        var wt = Array.prototype.map.call(table.querySelectorAll('tbody tr'), function (r) {
            var c = r.querySelectorAll('td');
            return { appt: (c[apptIdx] ? c[apptIdx].textContent.trim() : ''), hunter: (c[hunterIdx] ? c[hunterIdx].textContent.trim() : ''), submitted: (subIdx >= 0 && c[subIdx] ? c[subIdx].textContent.trim() : '') };
        }).filter(function (x) { return /\bWT\b/.test(x.appt) && x.hunter; });
        if (!wt.length) return null;
        wt.sort(function (a, b) { return parseDate(b.submitted) - parseDate(a.submitted); });
        return wt[0].hunter;
    }

    function dokuBridge() {
        // Poll until table has WT data, then postMessage to opener and close.
        var tries = 0;
        var poll = setInterval(function () {
            tries++;
            var hunter = null;
            try { hunter = readWebHunter(); } catch (e) {}
            if (hunter) {
                clearInterval(poll);
                try { if (window.opener) window.opener.postMessage({ type: 'WWWE_WT_HUNTER', hunter: hunter }, '*'); } catch (e) {}
                setTimeout(function () { try { window.close(); } catch (e) {} }, 300);
            } else if (tries > 60) {
                clearInterval(poll);
                try { if (window.opener) window.opener.postMessage({ type: 'WWWE_WT_HUNTER', hunter: null }, '*'); } catch (e) {}
                setTimeout(function () { try { window.close(); } catch (e) {} }, 300);
            }
        }, 500);
    }

    /* ===== 11) BOOT ===== */
    function boot() {
        if (isDoku()) { dokuBridge(); return; }
        if (readPayload()) { runZammad().catch(function (e) { console.warn('Zammad-Extension:', e); }); }
    }
    // Web-Hunter-Relay von der IPSI-Seite (Domaintransfer): Betreff nachtragen.
    window.addEventListener("message", function (ev) {
        var d = ev && ev.data;
        if (!d || d.type !== "WWWE_SET_HUNTER" || !d.hunter) return;
        var el = pickVisible('input[name="title"]');
        if (!el) return;
        var v = el.value || "";
        if (v.indexOf("Domaintransfer") < 0) return;
        var idx = v.indexOf("|");
        var base = idx >= 0 ? v.slice(0, idx + 1) : (v.replace(/\s*$/, "") + " |");
        setNativeValue(el, base + " " + String(d.hunter).trim());
    });
    window.addEventListener('hashchange', function () {
        if (!isDoku() && readPayload()) { runZammad().catch(function (e) { console.warn('Zammad-Extension:', e); }); }
    });
    if (document.readyState === 'complete' || document.readyState === 'interactive') { setTimeout(boot, 600); }
    else { window.addEventListener('DOMContentLoaded', function () { setTimeout(boot, 600); }); }
